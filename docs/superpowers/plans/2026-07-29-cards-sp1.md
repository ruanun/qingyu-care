```java
package com.qingyu.care.service;

import com.fasterxml.jackson.databind.ObjectMapper;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;

import com.qingyu.care.dto.UiPart;

import java.util.ArrayList;
import java.util.List;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

/**
 * 解析 AI 回复里内联的 ```spec 围栏为有序 uiParts。fail-soft：非法/未闭合围栏并入文本、绝不抛异常阻断回复。
 * 前端只渲染 uiParts、不含解析器（干净分层）。
 *
 * @author ruan
 */
@Service
@RequiredArgsConstructor
@Slf4j
public class UiSpecService {

    private final ObjectMapper objectMapper;

    /** 闭合的 ```spec ... ``` 围栏；DOTALL 让 . 匹配换行，非贪婪取最近的闭合。 */
    private static final Pattern FENCE = Pattern.compile("```spec\\s*\\r?\\n(.*?)```", Pattern.DOTALL);

    public record ParseResult(List<UiPart> uiParts, String plainText) {}

    public ParseResult parse(String raw) {
        if (raw == null || raw.isBlank()) {
            String s = raw == null ? "" : raw;
            return new ParseResult(List.of(UiPart.text(s)), s);
        }
        List<UiPart> parts = new ArrayList<>();
        StringBuilder plain = new StringBuilder();
        Matcher m = FENCE.matcher(raw);
        int last = 0;
        while (m.find()) {
            String before = raw.substring(last, m.start());
            if (!before.isBlank()) { parts.add(UiPart.text(before)); plain.append(before); }
            String json = m.group(1).trim();
            try {
                Object node = objectMapper.readValue(json, Object.class);
                parts.add(UiPart.ui(node));
            } catch (Exception e) {
                // fail-soft：非法 JSON 的围栏内容并入文本，不抛
                log.debug("卡片围栏 JSON 非法，回退为文本: {}", e.getMessage());
                parts.add(UiPart.text(json)); plain.append(json);
            }
            last = m.end();
        }
        String after = raw.substring(last);   // 含未闭合围栏残余 → 当文本，fail-soft
        if (!after.isBlank()) { parts.add(UiPart.text(after)); plain.append(after); }
        if (parts.isEmpty()) { parts.add(UiPart.text(raw)); plain.append(raw); }
        return new ParseResult(parts, plain.toString().trim());
    }
}
```

/** 仅允 http/https 且长度受限，其余返回空（防 javascript:/data: 注入）。 */
export function safeUrl(u: unknown): string {
  if (typeof u !== 'string' || u.length > 2048) return '';
  try { const p = new URL(u); return (p.protocol === 'http:' || p.protocol === 'https:') ? u : ''; }
  catch { return ''; }
}

- [ ] **Step 3：写 `JsonRender.tsx`（递归 + 未知 type 忽略）**

```tsx
import React from 'react';
import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
import { Button } from '@/components/ui/button';
import { Badge } from '@/components/ui/badge';
import { Separator } from '@/components/ui/separator';
import type { UiPart, UiNode } from '@/types';
import { safeUrl } from './registry';

function str(v: unknown): string { return v == null ? '' : String(v); }

function renderNode(node: UiNode, onAction: (v: string) => void, key: React.Key): React.ReactNode {
  const p = node.props ?? {};
  const kids = (node.children ?? []).map((c, i) => renderNode(c, onAction, i));
  switch (node.type) {
    case 'card':
      return <Card key={key} className="my-2">{p.title ? <CardHeader><CardTitle>{str(p.title)}</CardTitle></CardHeader> : null}<CardContent className="space-y-1">{kids}</CardContent></Card>;
    case 'text':
      return <div key={key} className={p.emphasis ? 'font-semibold' : ''}>{str(p.value ?? p.text)}</div>;
    case 'kv':
      return <div key={key} className="flex justify-between text-sm"><span className="text-muted-foreground">{str(p.label)}</span><span>{str(p.value)}</span></div>;
    case 'badge':
      return <Badge key={key} variant="secondary">{str(p.text ?? p.value)}</Badge>;
    case 'divider':
      return <Separator key={key} className="my-2" />;
    case 'image': {
      const src = safeUrl(p.src ?? p.url);
      return src ? <img key={key} src={src} alt={str(p.alt)} className="max-w-full rounded" /> : null;
    }
    case 'button':
      return <Button key={key} size="sm" variant="outline" className="mr-2 mt-1" onClick={() => onAction(str(p.value ?? p.label))}>{str(p.label)}</Button>;
    case 'actions':
      return <div key={key} className="flex flex-wrap">{kids}</div>;
    case 'list':
      return <ul key={key} className="list-disc pl-5">{kids}</ul>;
    case 'row':
      return <div key={key} className="flex gap-2">{kids}</div>;
    case 'column':
      return <div key={key} className="flex flex-col gap-1">{kids}</div>;
    default:
      return null;   // 未知 type 忽略，不报错
  }
}

/** 渲染一条 AI 回复的有序 uiParts：text 片段纯文本、ui 片段过组件注册表。 */
export function JsonRender({ parts, onAction }: { parts: UiPart[]; onAction: (v: string) => void }) {
  return (
    <div className="space-y-1">
      {parts.map((part, i) =>
        part.kind === 'text'
          ? (part.text ? <div key={i} className="whitespace-pre-wrap">{part.text}</div> : null)
          : (part.node ? <React.Fragment key={i}>{renderNode(part.node, onAction, i)}</React.Fragment> : null),
      )}
    </div>
  );
